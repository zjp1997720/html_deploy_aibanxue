# HTML代码分享项目 - 监控系统改进实施方案

## 📋 方案概述

本方案旨在为HTML代码分享项目建立完善的监控告警体系，提高系统的可观测性和故障响应能力。

## 🎯 实施目标

1. **监控覆盖度**: 从当前的40%提升到95%
2. **故障发现时间**: 从用户报告缩短到自动发现（<1分钟）
3. **MTTR**: 从数小时降低到30分钟以内
4. **系统可用性**: 提升到99.9%

```yaml
global:
  scrape_interval: 15s
  evaluation_interval: 15s

rule_files:
  - "alert_rules.yml"

scrape_configs:
  # Prometheus自身监控
  - job_name: 'prometheus'
    static_configs:
      - targets: ['localhost:9090']

  # Node Exporter - 系统指标
  - job_name: 'node'
    static_configs:
      - targets: ['localhost:9100']

  # 应用监控
  - job_name: 'html-go'
    static_configs:
      - targets: ['localhost:8888']

    scrape_interval: 10s

  # Blackbox Exporter - 端口监控
  - job_name: 'blackbox'
    metrics_path: /probe

      module: [http_2xx]
    static_configs:
      - targets:
        - http://localhost:8888/api/v2/health
    relabel_configs:
      - source_labels: [__address__]
        target_label: __param_target
      - source_labels: [__param_target]
        target_label: instance
      - target_label: __address__
        replacement: localhost:9115
```

#### 1.3 配置告警规则 (`alert_rules.yml`)
```yaml
groups:
  - name: system
    rules:
      # 服务不可用
      - alert: ServiceDown
        expr: up == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "服务 {{ $labels.instance }} 不可用"
          description: "{{ $labels.job }} 服务已经停止超过1分钟"

      # CPU使用率过高
      - alert: HighCPUUsage
        expr: 100 - (avg by(instance) (irate(node_cpu_seconds_total{mode="idle"}[5m])) * 100) > 80
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "{{ $labels.instance }} CPU使用率过高"
          description: "CPU使用率: {{ $value }}%"

      # 内存使用率过高
      - alert: HighMemoryUsage
        expr: (1 - (node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes)) * 100 > 85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "{{ $labels.instance }} 内存使用率过高"
          description: "内存使用率: {{ $value }}%"

      # 磁盘空间不足
      - alert: DiskSpaceLow
        expr: (1 - (node_filesystem_avail_bytes / node_filesystem_size_bytes)) * 100 > 85
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "{{ $labels.instance }} 磁盘空间不足"
          description: "磁盘 {{ $labels.mountpoint }} 使用率: {{ $value }}%"

  - name: application
    rules:
      # 应用错误率过高
      - alert: HighErrorRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) > 0.05
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "应用错误率过高"
          description: "5xx错误率: {{ $value | humanizePercentage }}"

      # 响应时间过长
      - alert: SlowResponse
        expr: histogram_quantile(0.95, rate(http_request_duration_seconds_bucket[5m])) > 2
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "响应时间过长"
          description: "95%分位响应时间: {{ $value }}秒"
```

### 第二阶段：应用监控改造（第2周）

#### 2.1 添加Prometheus指标导出
```javascript
// middleware/prometheusMetrics.js
const client = require('prom-client');

// 创建默认的注册表
const register = new client.Registry();

// 默认指标
client.collectDefaultMetrics({ register });

// 自定义指标
const httpRequestDuration = new client.Histogram({
  name: 'http_request_duration_seconds',
  help: 'HTTP请求持续时间',
  labelNames: ['method', 'route', 'status_code'],
  buckets: [0.1, 0.5, 1, 2, 5, 10]
});

const httpRequestTotal = new client.Counter({
  name: 'http_requests_total',
  help: 'HTTP请求总数',
  labelNames: ['method', 'route', 'status_code']
});

const activeConnections = new client.Gauge({
  name: 'http_active_connections',
  help: '当前活跃连接数'
});

const databaseOperations = new client.Histogram({
  name: 'database_operations_duration_seconds',
  help: '数据库操作持续时间',
  labelNames: ['operation', 'table'],
  buckets: [0.001, 0.01, 0.1, 1, 5]
});

// 导出中间件
function metricsMiddleware(req, res, next) {
  const start = Date.now();
  activeConnections.inc();
  
  res.on('finish', () => {
    const duration = (Date.now() - start) / 1000;
    const labels = {
      method: req.method,
      route: req.route?.path || req.path,
      status_code: res.statusCode
    };
    
    httpRequestDuration.observe(labels, duration);
    httpRequestTotal.inc(labels);
    activeConnections.dec();
  });
  
  next();
}

// 导出指标端点
function metricsEndpoint(req, res) {
  res.set('Content-Type', register.contentType);
  res.end(register.metrics());
}

module.exports = {
  metricsMiddleware,
  metricsEndpoint,
  register,
  httpRequestDuration,
  httpRequestTotal,
  activeConnections,
  databaseOperations
};
```

#### 2.2 集成到应用
```javascript
// 在app.js中添加
const { metricsMiddleware, metricsEndpoint } = require('./middleware/prometheusMetrics');

// 在其他中间件之后添加
app.use(metricsMiddleware);

// 添加指标端点
app.get('/metrics', metricsEndpoint);
```

### 第三阶段：日志管理（第3周）

#### 3.1 结构化日志
```javascript
// utils/logger.js
const winston = require('winston');
const { format, transports } = winston;

// 定义日志格式
const logFormat = format.combine(
  format.timestamp(),
  format.errors({ stack: true }),
  format.json()
);

// 创建logger实例
const logger = winston.createLogger({
  level: process.env.LOG_LEVEL || 'info',
  format: logFormat,
  defaultMeta: { service: 'html-go' },
  transports: [
    // 错误日志单独文件
    new transports.File({ 
      filename: 'logs/error.log', 
      level: 'error',
      maxsize: 5242880, // 5MB
      maxFiles: 5
    }),
    // 所有日志
    new transports.File({ 
      filename: 'logs/combined.log',
      maxsize: 5242880, // 5MB
      maxFiles: 10
    })
  ]
});

// 开发环境输出到控制台
if (process.env.NODE_ENV !== 'production') {
  logger.add(new transports.Console({
    format: format.simple()
  }));
}

module.exports = logger;
```

#### 4.2 告警脚本
```bash
#!/bin/bash
# scripts/send-alert.sh

# 接收参数
ALERT_NAME=$1
SEVERITY=$2
DESCRIPTION=$3

# 根据严重程度选择通知方式
case $SEVERITY in
  critical)
    # 紧急电话+短信+微信
    curl -X POST "https://api.example.com/phone" -d "message=$ALERT_NAME"
    curl -X POST "https://api.example.com/sms" -d "message=$ALERT_NAME"
    ;;
  warning)
    # 微信+邮件
    curl -X POST "https://api.example.com/wechat" -d "message=$ALERT_NAME"
    echo "$DESCRIPTION" | mail -s "[$SEVERITY] $ALERT_NAME" devops@example.com
    ;;
  info)
    # 仅邮件
    echo "$DESCRIPTION" | mail -s "[$SEVERITY] $ALERT_NAME" devops@example.com
    ;;
esac
```
